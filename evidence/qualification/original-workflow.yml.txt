name: qualified-sample-workflow
on: [push, pull_request, workflow_dispatch]
permissions:
  contents: read
jobs:
  qualified-linux:
    runs-on: ubuntu-24.04
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@11bd71901bbe5b1630ceea73d27597364c9af683 # v4.2.2
      - name: Obtain pinned qualified runtime
        run: |
          curl --fail --location --retry 3 'https://github.com/astral-sh/python-build-standalone/releases/download/20260924/cpython-3.12.14%2B20260924-x86_64-unknown-linux-gnu-install_only_stripped.tar.gz' -o runtime.tar.gz
          echo '269b2c99e4db15b242bf01832f4fea1e8f1a664f273cff519393f296e9820b41  runtime.tar.gz' | sha256sum --check
          mkdir runtime
          tar -xzf runtime.tar.gz -C runtime
          runtime/python/bin/python3 -m venv .venv
          .venv/bin/python -m pip install setuptools==84.0.0
      - name: Source tests and independent semantic oracle
        run: |
          PYTHONPATH=src .venv/bin/python -m unittest discover -s tests -v
          PYTHONPATH=src .venv/bin/python qualification/independent_semantics.py --output independent-parity.json
          PYTHONPATH=src .venv/bin/python qualification/calibrate_limits.py --output cap-calibration.json
      - name: Build, install, and exercise the README
        run: |
          .venv/bin/python qualification/build_dist.py
          .venv/bin/python -m pip install --no-deps dist/*.whl
          .venv/bin/sqlitefolio profile
          .venv/bin/sqlitefolio rehearse examples/rebuild/manifest.json --out review --trust-input
          .venv/bin/sqlitefolio verify review
          .venv/bin/sqlitefolio inspect review > review-copy.html
          .venv/bin/sqlitefolio replay review --manifest examples/rebuild/manifest.json --trust-input
          env -u PYTHONPATH .venv/bin/python -m unittest discover -s tests -v
